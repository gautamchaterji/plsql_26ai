CREATE TABLE BRICKS2
    AS
        SELECT
            *
        FROM
            BRICKS
        WHERE
            1 = 0;

CREATE SEQUENCE BRICKS_SEQ START WITH 1 INCREMENT BY 1;

CREATE OR REPLACE PROCEDURE INSERT_BRICKS (
    P_COLOUR BRICKS2.COLOUR%TYPE,
    P_SHAPE  BRICKS2.SHAPE%TYPE,
    P_WEIGHT BRICKS2.WEIGHT%TYPE
) IS
BEGIN
    INSERT INTO BRICKS2 (
        BRICK_ID,
        COLOUR,
        SHAPE,
        WEIGHT
    ) VALUES ( BRICKS_SEQ.NEXTVAL,
               P_COLOUR,
               P_SHAPE,
               P_WEIGHT );

    COMMIT;
EXCEPTION
    WHEN OTHERS THEN
        DBMS_OUTPUT.PUT_LINE('Insert statement failed!!');
END;

CREATE OR REPLACE PROCEDURE BRICKS_SELECT (
    P_BRICK_ID BRICKS.BRICK_ID%TYPE
) IS
    R_BRICKS BRICKS%ROWTYPE;
BEGIN
    SELECT
        A.*
    INTO R_BRICKS
    FROM
        BRICKS A
    WHERE
        A.BRICK_ID = P_BRICK_ID;

    INSERT_BRICKS(R_BRICKS.COLOUR, R_BRICKS.SHAPE, R_BRICKS.WEIGHT);
EXCEPTION
    WHEN NO_DATA_FOUND THEN
        DBMS_OUTPUT.PUT_LINE('No data found!!!');
    WHEN OTHERS THEN
        DBMS_OUTPUT.PUT_LINE('Other error!!!');
END;

SELECT
    BRICK_ID,

SELECT
    BRICK_ID,
    COLOUR,
    SHAPE,
    WEIGHT

BEGIN
    BRICKS_SELECT(1);
END;
